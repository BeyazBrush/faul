 1. IMPORT ZİNCİRİ  (l0 — ana menü, DontDestroyOnLoad)
--------------------------------------------------------------------------------
Bu zincirin TAMAMINI CdMusicImport.cs biliyor. Değişken adları asset'ten
doğrulandı, tahmin yok.

  Radio/CD         GO "CD"  FSM "Playlist"    (start: Check)
      Check   → DontDestroyOnLoad + GetOwner → storeGameObject = SongDatabaseCD
      Init    → GameObjectCompare(SongDatabaseCD) → DestroySelf → State 3
      State 3 → ActivateGameObject ×3 + BoolAllTrue → State 1
      State 1 → SetStringValue  Path = "NOT IMPORTED" / "CD'S IMPORTED"
      olaylar : ADVERT1 DESTROY FAIL JINGLE LOAD LOOP "NEXT SONG" PLAY
                "PREVIOUS SONG" RADIO_FOLK WAKEUP CD1 CD2 CD3 Import

  Radio/CD/CD1     GO "CD1" FSM "LoadSongs"  (start: Init)   [ayni CD2, CD3]
      Init          ArrayListClear("Ref") + Wait + SetFsmString
      Get file name GetAppPath → DataPathResult / AppFolderResult /
                    PersistentPathResult / PlaformResult
                    IntAdd(Name,+1) → ConvertIntToString → BuildString(
                      "file:///" + <appPath> + FolderName + "track" + Name + ".ogg"
                      → Path)
                    IntCompare(Name, MaxTracks): lessThan→Load, equal→Off
      Load          WWWPOST(url=Path, storeAudio=Song, errorString=Error)
                    error→"Fail", başarı→"Add List"
      Add List      ArrayListAdd("Ref", Song)      ← ASIL IMPORT
      Off           SetFsmBool + SetIntValue + EnableFSM
      değişkenler  : MaxTracks(int) Track(int) Error(str) FolderName(str)
                    Name(str) Path(str) Ref(list) Song(AudioClip)
      MaxTracks = 16  → BU SLOT SAYISI, yüklenen şarkı sayısı DEĞİL

  Streaming        GO "Streaming" FSM "Kansanradio"   (radyo kanalları için
                    Playlist'in birebir aynısı; SongDatabase + "file:///Radio/")

  Radio/Folk       GO "Folk" FSM "LoadSongs"  +  FSM "Kansanradio"
